package main

import (
	"fmt"
	"log"
	"net/http"
	"os" // lettura delle variabili
	"path/filepath"

	"github.com/adrianopiccinini/roccia-fit-api/pkg/controllers"
	"github.com/adrianopiccinini/roccia-fit-api/pkg/database"
	"github.com/adrianopiccinini/roccia-fit-api/pkg/middleware"
	"github.com/adrianopiccinini/roccia-fit-api/pkg/models"
	httpSwagger "github.com/swaggo/http-swagger"

	_ "github.com/adrianopiccinini/roccia-fit-api/docs"
	"github.com/go-chi/chi/v5"
	chiMiddleware "github.com/go-chi/chi/v5/middleware"
	"github.com/go-chi/cors"
	"github.com/joho/godotenv" // Metodo per caricare il file .env
)

// @title RocciaFit API
// @version 1.0
// @description API per gestione la gestione dei parametri d'allenamento e condizione fisica
// @termsOfService http://swagger.io/terms/
// @host localhost
// @BasePath /api/v1
// @securityDefinitions.apikey BearerAuth
// @in header
// @name Authorzation

func main() {

	if err := godotenv.Load(); err != nil {
		log.Println("Nessun file .env trovato, verrano  usate le variabili di sistema")
	}

	dsn := os.Getenv("DB_DSN")

	if dsn == "" {
		log.Fatal("DB_DSN non impostato nelle variabili d'ambiente")
	}

	//inizializziamo la connessione al db
	db := database.InitDB(dsn)

	//defer mette in coda questa istruzione istruzione: chiudere la connesione
	// che verra attivata solo quando chiudero il server go
	// Estraiamo l'oggetto SQL generico per poter usare il defer Close
	sqlDB, err := db.DB()
	if err == nil {
		defer sqlDB.Close()
	}

	// AUTO-MIGRAZIONE
	//GORM legge le struct e crea la tabella su PostgreSQL
	fmt.Println("Esecuzione della migrazione del database...")
	if err := db.AutoMigrate(
		&models.Utente{},
		&models.SchedaAllenamento{},
		&models.GiornoAllenamento{},
		&models.EsercizioScheda{},
		&models.Esercizio{},
		&models.Allenamento{},
		&models.SerieEseguita{},
		&models.PesoCorporeo{},
		&models.CheckFoto{},
		&models.Dieta{},
		&models.GiornoDieta{},
		&models.Pasto{},
		&models.AlimentoPasto{},
		&models.AlternativaAlimento{},
		&models.ProtocolloCardio{},
		&models.SessioneCardio{},
		&models.Feedback{},
		&models.CheckIn{},
		&models.Messaggio{},
	); err != nil {
		log.Fatalf("Errore durante la migrazione: %v", err) // %v e' un segnaposto usato per stampare log
	}
	fmt.Println("Migrazione completata con successo!")

	utenteController := &controllers.UtenteController{DB: db}
	schedeController := &controllers.SchedaController{DB: db}
	esercizioController := &controllers.EsercizioController{DB: db}
	allenamentoController := &controllers.AllenamentoController{DB: db}
	pesoController := &controllers.PesoController{DB: db}
	checkFotoController := &controllers.CheckFotoController{DB: db}
	authController := &controllers.AuthController{DB: db}
	statsController := &controllers.StatsController{DB: db}
	dietaController := &controllers.DietaController{DB: db}
	cardioController := &controllers.CardioController{DB: db}
	feedbackController := &controllers.FeedbackController{DB: db}
	messaggioController := &controllers.MessaggioController{DB: db}
	chatAIController := &controllers.ChatAIController{DB: db}

	// Creiamo il ROUTER chi
	r := chi.NewRouter()

	// Leggi i domini permessi dal file .env, altrimenti usa "*" solo come fallback di sviluppo
	allowedOrigin := os.Getenv("ALLOWED_ORIGIN")
	if allowedOrigin == "" {
		allowedOrigin = "*"
	}

	// CONFIGURAZIONE PER L'APP MOBILE
	r.Use(cors.Handler(cors.Options{
		AllowedOrigins:   []string{allowedOrigin},
		AllowedMethods:   []string{"GET", "POST", "PUT", "DELETE", "OPTIONS"},
		AllowedHeaders:   []string{"Accept", "Authorization", "Content-Type", "X-CSRF-Token"},
		ExposedHeaders:   []string{"Link"},
		AllowCredentials: true,
		MaxAge:           300,
	}))

	r.Use(chiMiddleware.Logger)    // stampa a terminale ogni richiesta HTTP
	r.Use(chiMiddleware.Recoverer) // Impedisce al server di crashare se ce un errore grave

	// ESPOSIZIONE DEI FILE STATICI: permette di visualizzare le foto tramite URL
	workDir, _ := os.Getwd()
	filesDir := http.Dir(filepath.Join(workDir, "uploads"))
	r.Handle("/uploads/*", http.StripPrefix("/uploads", http.FileServer(filesDir)))

	r.Get("/health", func(w http.ResponseWriter, req *http.Request) {
		w.Header().Set("Content-Type", "application/json")
		w.WriteHeader(http.StatusOK)
		w.Write([]byte(`{"status": "dio bessia" , "message": "API operative"}`))
	})

	r.Get("/swagger/*", httpSwagger.WrapHandler)

	r.Route("/api/v1", func(r chi.Router) {
		r.Route("/auth", func(r chi.Router) {
			r.Post("/registrati", authController.RegistrazioneSicura)
			r.Post("/login", authController.Login)
		})

		r.Group(func(r chi.Router) {
			//applica il middleware a tutto quello che si trova nel middleware
			r.Use(middleware.AuthMiddleware)

			r.Route("/utenti", func(r chi.Router) {
				r.Post("/", utenteController.CreaUtente)
				r.Post("/bulk", utenteController.CreaUtentiMultipli)
				r.Get("/", utenteController.GetUtenti)
				r.Get("/classifica", utenteController.GetClassifica)

				r.Route("/{id}", func(r chi.Router) {
					r.Get("/", utenteController.GetUtenteByID)
					r.Put("/", utenteController.UpdateUtente)
					r.Delete("/", utenteController.DeleteUtente)
					r.Post("/xp", utenteController.AggiungiXP)
					r.Post("/checkin", utenteController.GestisciCheckIn)
					r.Post("/acqua", utenteController.GestisciAcqua)
					r.Post("/push-token", utenteController.SalvaPushToken)
				})
			})
			r.Route("/schede", func(r chi.Router) {
				r.Get("/test", func(w http.ResponseWriter, req *http.Request) {
					w.Write([]byte("Ecco le schede di allenamento"))
				})

				r.Post("/", schedeController.CreaScheda)
				r.Get("/atleta/{idAtleta}", schedeController.GetSchedeAtleta)

				r.Route("/{id}", func(r chi.Router) {
					r.Put("/", schedeController.UpdateScheda)
					r.Get("/", schedeController.GetSchedaByID)
					r.Delete("/", schedeController.DeleteScheda)
				})
			})
			r.Route("/esercizi", func(r chi.Router) {
				r.Get("/", esercizioController.GetEsercizi)
				r.Post("/", esercizioController.CreaEsercizio)
				r.Post("/bulk", esercizioController.CreaEserciziMultipli)
				r.Route("/{id}", func(r chi.Router) {
					r.Get("/", esercizioController.GetEsercizioByID)
					r.Put("/", esercizioController.UpdateEsercizio)
					r.Delete("/", esercizioController.DeleteEsercizio)
				})
			})
			r.Route("/allenamenti", func(r chi.Router) {

				r.Post("/inizia", allenamentoController.IniziaAllenamento)
				r.Get("/atleta/{idAtleta}", allenamentoController.GetAllenamentiAtleta)

				r.Route("/serie", func(r chi.Router) {
					r.Post("/", allenamentoController.AggiungiSerie)
					r.Put("/{id}", allenamentoController.UpdateSerie)
					r.Delete("/{id}", allenamentoController.DeleteSerie)
				})

				r.Route("/{id}", func(r chi.Router) {
					r.Get("/", allenamentoController.GetAllenamentoById)
					r.Delete("/", allenamentoController.DeleteAllenamento)
				})
			})
			r.Route("/peso", func(r chi.Router) {
				r.Post("/", pesoController.AggiungiPeso)
				r.Post("/bulk", pesoController.AggiungiPesiMultipli)
				r.Get("/atleta/{idAtleta}", pesoController.GetStoricoPeso)
				r.Get("/ultimo/{idAtleta}", pesoController.GetUltimoPeso)

				r.Route("/{id}", func(r chi.Router) {
					r.Get("/", pesoController.GetPesoByID)
					r.Put("/", pesoController.UpdatePeso)
					r.Delete("/", pesoController.DeletePeso)
				})
			})
			r.Route("/check", func(r chi.Router) {
				r.Post("/", checkFotoController.AggiungiCheckFoto)
				r.Post("/upload", checkFotoController.UploadImmagine)
				r.Get("/atleta/{idAtleta}", checkFotoController.GetCheckAtleta)

				r.Route("/{id}", func(r chi.Router) {
					r.Get("/", checkFotoController.GetCheckFotoByID)
					r.Put("/", checkFotoController.UpdateCheckFoto)
					r.Delete("/", checkFotoController.DeleteCheckFoto)
				})
			})
			r.Route("/stats", func(r chi.Router) {
				r.Get("/tonnellaggio/{idAllenamento}", statsController.GetTonnellaggio)
				r.Get("/massimale/{idSerie}", statsController.GetProiezione1RM)
				r.Get("/volume/{idAtleta}", statsController.GetVolumeSettimanale)
			})
			r.Route("/dieta", func(r chi.Router) {
				r.Get("/atleta/{idAtleta}", dietaController.GetDietaAttivaAtleta)
				r.Get("/atleta/{idAtleta}/all", dietaController.GetDieteAtleta)
				r.Post("/", dietaController.CreaDieta)

				r.Route("/{id}", func(r chi.Router) {
					r.Get("/", dietaController.GetDietaByID)
					r.Put("/", dietaController.UpdateDieta)
					r.Delete("/", dietaController.DeleteDieta)
				})
			})
			r.Route("/storico", func(r chi.Router) {
				r.Get("/atleta/{id}", utenteController.GetStoricoAtleta)
			})
			r.Route("/chat", func(r chi.Router) {
				r.Post("/", messaggioController.InviaMessaggio)
				r.Get("/{id1}/{id2}", messaggioController.GetCronologiaChat)
				r.Get("/ws/{id}", messaggioController.ConnettiWS)
				r.Post("/ai", chatAIController.GeneraRispostaAI)
			})

			// --- ROTTE CARDIO ---
			r.Get("/cardio/atleta/{idAtleta}", cardioController.GetCardioAtleta)
			r.Post("/cardio", cardioController.SalvaCardio)
			r.Post("/cardio/", cardioController.SalvaCardio)

			// --- ROTTE FEEDBACK ---
			r.Get("/feedback/atleta/{idAtleta}", feedbackController.GetFeedbackAtleta)
			r.Post("/feedback", feedbackController.CreaFeedback)
			r.Post("/feedback/", feedbackController.CreaFeedback)

			// --- CAMBIO PASSWORD ---
			r.Put("/auth/password", authController.CambiaPassword)
		})
	})

	// qui caricheremo le variabili d'ambiente
	port := os.Getenv("PORT")
	if port == "" {
		port = ":8080"
	} else if port[0] != ':' {
		port = ":" + port // Se Render ci passa "10000", noi lo trasformiamo in ":10000"
	}

	fmt.Printf("Server in ascolto sulla porta %s...\n", port)

	// accendiamo il server
	if err := http.ListenAndServe(port, r); err != nil {
		log.Fatalf("Errore durante l'avvio del server: %v", err)
	}
}

// GetAllenamentiAtleta godoc
// @Summary Storico allenamenti
// @Description Recupera gli allenamenti in formato paginato
// @Tags Allenamenti
// @Produce json
// @Param idAtleta path string true "ID Atleta"
// @Param page query int false "Numero pagina (default: 1)"
// @Param limit query int false "Elementi per pagina (default: 10)"
// @Security BearerAuth
// @Success 200 {object} map[string]interface{}
// @Router /allenamenti/atleta/{idAtleta} [get]
func (c *AllenamentoController) GetAllenamentiAtleta(w http.ResponseWriter, r *http.Request) {
	// GET /atleti/123/allenamenti?page=2&limit=10
	// 123 e' un PATH PARAMETER
	// mentre ?page=2&limit=10 sono dei QUERY PARAMETERS
	idAtleta := chi.URLParam(r, "idAtleta")

	// Lettura dei query parameters con valori di default di sicurezza, prendono delle stringhe non dei numeri
	pageStr := r.URL.Query().Get("page")
	limitStr := r.URL.Query().Get("limit")

	page := 1
	limit := 10

	// Li convertiamo in numeri perche inizalmente erano stringhe Atoi (ASCII to Integer)
	// p -> numero convertito 	err -> eventuale errore
	if p, err := strconv.Atoi(pageStr); err == nil && p > 0 {
		page = p
	}

	if l, err := strconv.Atoi(limitStr); err == nil && l > 0 {
		limit = l
	}

	// offset indica da quale record iniziare la lettura
	offset := (page - 1) * limit

	var allenamenti []models.Allenamento
	var totaleRecord int64

	// Conteggio totale dei record , serve per dire al front end che queste pagine esistono
	//? -> permette a GORM/database driver di gestire correttamente il valore e aiuta a evitare SQL injection.
	c.DB.Model(&models.Allenamento{}).Where("idAtleta = ?", idAtleta).Count(&totaleRecord)

	risultato := c.DB.Preload("SerieEseguite").
		Where("id_atleta = ?", idAtleta).
		Order("data desc").
		Limit(limit).
		Offset(offset).
		Find(&allenamenti)

	if risultato.Error != nil {
		http.Error(w, "Errore durante il recupero degli allenamenti", http.StatusInternalServerError)
		return
	}

	// Creazione dell'envelope di risposta , l'envelope restituisce i dati all'interno di una struttura invece di restituire direttamente i dati
	risposta := map[string]interface{}{
		"dati":          allenamenti,
		"pagina":        page,
		"limite":        limit,
		"totale":        totaleRecord,
		"totale_pagine": (int(totaleRecord) + limit - 1) / limit,
	}

	w.Header().Set("Content-Type", "application/json")
	w.WriteHeader(http.StatusOK)
	json.NewEncoder(w).Encode(risposta)

	/*
		// Ordiniamo per data decrescente così hai sempre l'ultimo allenamento in cima

package controllers

import (
	"encoding/json" // serve a lavorare con i json
	"net/http"      // crea il server http gestendo post, get, put e delete
	"os"            // permette di leggere le variabili d'ambiente
	"time"          // date e orari

	"github.com/adrianopiccinini/roccia-fit-api/pkg/models"
	"github.com/golang-jwt/jwt/v5"
	"github.com/google/uuid"
	"golang.org/x/crypto/bcrypt"
	"gorm.io/gorm"
)

type AuthController struct {
	DB *gorm.DB
}

// Struttura per lettura dati login
type Credenziali struct {
	Email    string `json:"email"`
	Password string `json:"password"`
}

type CambioPswReq struct {
	Email      string `json:"email"`
	VecchiaPsw string `json:"vecchia_psw"`
	NuovaPsw   string `json:"nuova_psw"`
}

// Struttura del Token JWT (carta d'identita contenuta nel token) Json Web Token
type Claims struct {
	UtenteID string `json:"utente_id"`
	jwt.RegisteredClaims
}

// RegistrazioneSicura godoc
// @Summary Registrazione nuovo utente
// @Description Crea un account con password crittografata
// @Tags Autenticazione
// @Accept json
// @Produce json
// @Param utente body models.Utente true "Dati utente"
// @Success 201 {object} models.Utente
// @Failure 400 {string} string "Errore nella decodifica"
// @Router /auth/registrati [post]

func (c *AuthController) RegistrazioneSicura(w http.ResponseWriter, r *http.Request) {
	var nuovoUtente models.Utente

	if err := json.NewDecoder(r.Body).Decode(&nuovoUtente); err != nil {
		http.Error(w, "Errore nella decodifica", http.StatusBadRequest)
		return
	}

	nuovoUtente.CF = uuid.New().String()

	// Bcrypt prende la password in chiaro e genera l'hash irreversibile
	hash, err := bcrypt.GenerateFromPassword([]byte(nuovoUtente.Psw), 10)
	if err != nil {
		http.Error(w, "Errore nella crittografia della password", http.StatusInternalServerError)
		return
	}

	nuovoUtente.ID = uuid.New().String()
	nuovoUtente.Psw = string(hash) //Salviamo l'hash nel db non la password

	if risultato := c.DB.Create(&nuovoUtente); risultato.Error != nil {
		http.Error(w, "Errore durante la creazione dell'utente", http.StatusInternalServerError)
		return
	}

	nuovoUtente.Psw = ""

	w.Header().Set("Content-Type", "application/json")
	w.WriteHeader(http.StatusCreated)
	json.NewEncoder(w).Encode(nuovoUtente)
}

// Login godoc
// @Summary Autenticazione utente
// @Description Verifica email e password e restituisce un token JWT
// @Tags Autenticazione
// @Accept json
// @Produce json
// @Param credenziali body Credenziali true "Dati di accesso"
// @Success 200 {object} map[string]interface{}
// @Failure 400 {string} string "Richiesta non valida"
// @Failure 401 {string} string "Credenziali errate"
// @Router /auth/login [post]
func (c *AuthController) Login(w http.ResponseWriter, r *http.Request) {
	var creds Credenziali
	if err := json.NewDecoder(r.Body).Decode(&creds); err != nil {
		http.Error(w, "Richiesta non valida", http.StatusBadRequest)
		return
	}

	// --- NUOVO CONTROLLO DI SICUREZZA ---
	// Blocchiamo immediatamente chi tenta di accedere con campi vuoti
	if creds.Email == "" || creds.Password == "" {
		http.Error(w, "Email e password sono obbligatori", http.StatusBadRequest)
		return
	}
	// ------------------------------------

	var utente models.Utente
	// 1. Cerchiamo l'utente tramite l'email
	if err := c.DB.Where("email = ?", creds.Email).First(&utente).Error; err != nil {
		http.Error(w, "Credenziali errate", http.StatusUnauthorized)
		return
	}

	// 2. Confrontiamo la password inserita con l'hash salvato nel database
	if err := bcrypt.CompareHashAndPassword([]byte(utente.Psw), []byte(creds.Password)); err != nil {
		http.Error(w, "Credenziali Errate", http.StatusUnauthorized)
		return
	}

	// Applichiamo una scadenza a tale token e la inseriamo in claims dove ci sono tutti i dati NON CIFRATI
	scadenza := time.Now().Add(7 * 24 * time.Hour)
	claims := &Claims{
		UtenteID: utente.ID,
		RegisteredClaims: jwt.RegisteredClaims{
			ExpiresAt: jwt.NewNumericDate(scadenza),
		},
	}

	token := jwt.NewWithClaims(jwt.SigningMethodHS256, claims)

	//JWT_SECRET e' una chiave che conosce solo il mio backend
	chiaveSegreta := os.Getenv("JWT_SECRET")
	tokenString, err := token.SignedString([]byte(chiaveSegreta))

	if err != nil {
		http.Error(w, "Errore nella generazione del token", http.StatusInternalServerError)
		return
	}

	w.Header().Set("Content-Type", "application/json")
	json.NewEncoder(w).Encode(map[string]interface{}{
		"token": tokenString,
		"utente": map[string]string{
			"id":    utente.ID,
			"nome":  utente.Nome,
			"ruolo": utente.Ruolo,
		},
	})
}

// CambiaPassword godoc
// @Summary Modifica password
// @Description Aggiorna la password verificando la precedente
// @Tags Autenticazione
// @Accept json
// @Produce json
// @Param credenziali body CambioPswReq true "Dati cambio password"
// @Security BearerAuth
// @Success 200 {string} string "Password aggiornata con successo"
// @Router /auth/password [put]
func (c *AuthController) CambiaPassword(w http.ResponseWriter, r *http.Request) {
	var req CambioPswReq
	if err := json.NewDecoder(r.Body).Decode(&req); err != nil {
		http.Error(w, "Richiesta non valida", http.StatusBadRequest)
		return
	}

	var utente models.Utente
	if err := c.DB.Where("email = ?", req.Email).First(&utente).Error; err != nil {
		http.Error(w, "Utente non trovato", http.StatusNotFound)
		return
	}

	if err := bcrypt.CompareHashAndPassword([]byte(utente.Psw), []byte(req.VecchiaPsw)); err != nil {
		http.Error(w, "La vecchia password non e' valida", http.StatusUnauthorized)
		return
	}

	nuovoHash, err := bcrypt.GenerateFromPassword([]byte(req.NuovaPsw), 10)
	if err != nil {
		http.Error(w, "Errore durante l'aggiornamento della password", http.StatusInternalServerError)
		return
	}

	utente.Psw = string(nuovoHash)
	if err := c.DB.Save(&utente).Error; err != nil {
		http.Error(w, "Errore durante l'aggironamento della password", http.StatusInternalServerError)
		return
	}

	w.Header().Set("Content-Type", "application/json")
	w.WriteHeader(http.StatusOK)
	w.Write([]byte(`{"message": "Password aggiornata con successo"}`))
}

// Salva o sovrascrive il protocollo Cardio dell'atleta
func (c *CardioController) SalvaCardio(w http.ResponseWriter, r *http.Request) {
	var input models.ProtocolloCardio
	if err := json.NewDecoder(r.Body).Decode(&input); err != nil {
		http.Error(w, "JSON non valido", http.StatusBadRequest)
		return
	}

	// Usiamo una transazione: eliminiamo il vecchio protocollo (se esiste) e salviamo il nuovo
	err := c.DB.Transaction(func(tx *gorm.DB) error {
		var existing models.ProtocolloCardio
		if err := tx.Where("id_atleta = ?", input.IDAtleta).First(&existing).Error; err == nil {
			// Se esiste un protocollo precedente, lo cancelliamo (le sessioni verranno rimosse a cascata)
			tx.Delete(&existing)
		}

		if err := tx.Create(&input).Error; err != nil {
			return err
		}
		return nil
	})

	if err != nil {
		http.Error(w, "Errore salvataggio cardio: "+err.Error(), http.StatusInternalServerError)
		return
	}

	w.Header().Set("Content-Type", "application/json")
	w.WriteHeader(http.StatusCreated)

// Struttura per ricevere la richiesta dall'app mobile
type AIRequest struct {
	IDAtleta  string `json:"id_atleta"`
	Messaggio string `json:"messaggio"`
}

// Strutture per comunicare con l'API di OpenAI
type OpenAIMessage struct {
	Role    string `json:"role"`
	Content string `json:"content"`
}

type OpenAIRequest struct {
	Model    string          `json:"model"`
	Messages []OpenAIMessage `json:"messages"`
}

type OpenAIResponse struct {
	Choices []struct {
		Message OpenAIMessage `json:"message"`
	} `json:"choices"`
}

// GeneraRispostaAI riceve la domanda, inietta il contesto fisico e chiama l'AI
func (c *ChatAIController) GeneraRispostaAI(w http.ResponseWriter, r *http.Request) {
	var req AIRequest
	if err := json.NewDecoder(r.Body).Decode(&req); err != nil {
		http.Error(w, "Richiesta JSON non valida", http.StatusBadRequest)
		return
	}

	// 1. RECUPERO IL CONTESTO DELL'ATLETA DAL DATABASE
	var utente models.Utente
	c.DB.First(&utente, "id = ?", req.IDAtleta)

	var ultimaPesata models.PesoCorporeo
	c.DB.Where("id_atleta = ?", req.IDAtleta).Order("data desc").First(&ultimaPesata)

	var dietaAttiva models.Dieta
	c.DB.Preload("GiorniDieta").Where("id_atleta = ? AND attiva = true", req.IDAtleta).First(&dietaAttiva)

	kcal := 0
	if len(dietaAttiva.GiorniDieta) > 0 {
		kcal = dietaAttiva.GiorniDieta[0].TargetKcal
	}

	// 2. CREAZIONE DEL SYSTEM PROMPT INVISIBILE (La Magia)
	systemPrompt := fmt.Sprintf(`Sei il Coach virtuale AI dell'app RocciaFit. 
	Parli in modo diretto, motivante e tecnico (stile bodybuilding). 
	Ecco i dati attuali dell'atleta con cui stai parlando:
	- Nome: %s
	- Categoria: %s
	- Ultimo peso registrato: %.1f kg
	- Calorie giornaliere target: %d kcal.
	Rispondi alla sua domanda basandoti su questi dati in modo conciso, senza superare le 4-5 righe.`,
		utente.Nome, utente.Categoria, ultimaPesata.Peso, kcal)

	// 3. PREPARAZIONE DEL PAYLOAD PER OPENAI
	openAIReq := OpenAIRequest{
		Model: "gpt-4o-mini", // Il modello economico ma intelligente
		Messages: []OpenAIMessage{
			{Role: "system", Content: systemPrompt},
			{Role: "user", Content: req.Messaggio},
		},
	}

	bodyBytes, _ := json.Marshal(openAIReq)

	// 4. CHIAMATA HTTP ALL'API DELL'AI
	apiKey := os.Getenv("OPENAI_API_KEY")
	if apiKey == "" {
		http.Error(w, "API Key mancante nel server", http.StatusInternalServerError)
		return
	}

	httpReq, _ := http.NewRequest("POST", "https://api.openai.com/v1/chat/completions", bytes.NewBuffer(bodyBytes))
	httpReq.Header.Set("Content-Type", "application/json")
	httpReq.Header.Set("Authorization", "Bearer "+apiKey)

	client := &http.Client{}
	resp, err := client.Do(httpReq)
	if err != nil || resp.StatusCode != 200 {
		http.Error(w, "Errore di comunicazione con l'intelligenza artificiale", http.StatusInternalServerError)
		return
	}
	defer resp.Body.Close()

	// 5. DECODIFICA DELLA RISPOSTA
	bodyData, _ := io.ReadAll(resp.Body)
	var openAIResp OpenAIResponse
	json.Unmarshal(bodyData, &openAIResp)

	rispostaFinale := "Scusa Roccia, non sono riuscito a elaborare il pensiero."
	if len(openAIResp.Choices) > 0 {
		rispostaFinale = openAIResp.Choices[0].Message.Content
	}

	// 6. INVIO AL FRONTEND
	w.Header().Set("Content-Type", "application/json")
	json.NewEncoder(w).Encode(map[string]string{
		"risposta": rispostaFinale,
	})
}

// UploadImmagine godoc
// @Summary Carica foto condizione
// @Description Salva il file fisico della foto di check (Max 10MB)
// @Tags Check Fotografici
// @Accept multipart/form-data
// @Produce json
// @Param foto formData file true "File immagine"
// @Security BearerAuth
// @Success 201 {object} map[string]string
// @Router /check/upload [post]
func (c *CheckFotoController) UploadImmagine(w http.ResponseWriter, r *http.Request) {
	// Limitiamo la dimensione della richiesta (es. 10 MB massimo)
	err := r.ParseMultipartForm(10 << 20)
	if err != nil {
		http.Error(w, "File troppo grande o richiesta non valida", http.StatusBadRequest)
		return
	}

	// Estraiamo il file dalla richiesta e si chiamera foto
	file, handler, err := r.FormFile("foto")
	if err != nil {
		http.Error(w, "Errore nel recupero del file", http.StatusBadRequest)
		return
	}
	defer file.Close()

	//Creazione della cartella degli upload in caso non ci sia
	uploadDir := "./uploads/checks"
	if _, err := os.Stat(uploadDir); os.IsNotExist(err) {
		os.MkdirAll(uploadDir, os.ModePerm)
	}

	// Generiamo un nome univoco uuid + estensione
	estensione := filepath.Ext(handler.Filename)
	nuovoNome := uuid.New().String() + estensione
	percorsoCompleto := filepath.Join(uploadDir, nuovoNome)

	// Creiamo il file fisico sul server
	dst, err := os.Create(percorsoCompleto)
	if err != nil {
		http.Error(w, "Errore nella creazione del file sul server", http.StatusInternalServerError)
		return
	}
	defer dst.Close()

	// Copiamo i byte caricati nel file del server
	if _, err := io.Copy(dst, file); err != nil {
		http.Error(w, "Errore durante il salvataggio dei dati", http.StatusInternalServerError)
		return
	}

	// Restituiamo il percorso relativo da salvare nel database
	urlFile := "/uploads/checks/" + nuovoNome

	w.Header().Set("Content-Type", "application/json")
	w.WriteHeader(http.StatusCreated)
	json.NewEncoder(w).Encode(map[string]string{"url_foto": urlFile})
}

// Funzione di utilità per inviare notifiche push tramite l'API di Expo
func InviaNotificaExpo(pushToken string, titolo string, corpo string) {
	if pushToken == "" {
		return
	}

	payload := map[string]interface{}{
		"to":    pushToken,
		"sound": "default",
		"title": titolo,
		"body":  corpo,
	}

	bodyBytes, _ := json.Marshal(payload)

	// Chiamata HTTP verso il servizio di Expo Push
	resp, err := http.Post("https://exp.host/--/api/v2/push/send", "application/json", bytes.NewBuffer(bodyBytes))
	if err != nil {
		return
	}
	defer resp.Body.Close()
}

	}

type MessaggioController struct {
	DB *gorm.DB
}

// Strutture WebSocket
var upgrader = websocket.Upgrader{
	CheckOrigin: func(r *http.Request) bool { return true },
}

var clients = make(map[string]*websocket.Conn)
var clientsMu sync.Mutex

// ⚡ 1. Rotta per inizializzare il WebSocket
func (c *MessaggioController) ConnettiWS(w http.ResponseWriter, r *http.Request) {
	idUtente := chi.URLParam(r, "id")

	conn, err := upgrader.Upgrade(w, r, nil)
	if err != nil {
		return
	}

	clientsMu.Lock()
	clients[idUtente] = conn
	clientsMu.Unlock()

	go func() {
		defer func() {
			clientsMu.Lock()
			delete(clients, idUtente)
			clientsMu.Unlock()
			conn.Close()
		}()
		for {
			if _, _, err := conn.ReadMessage(); err != nil {
				break
			}
		}
	}()
}

// ⚡ 2. Invio e salvataggio
func (c *MessaggioController) InviaMessaggio(w http.ResponseWriter, r *http.Request) {
	var messaggio models.Messaggio
	if err := json.NewDecoder(r.Body).Decode(&messaggio); err != nil {
		http.Error(w, "Payload invalido", http.StatusBadRequest)
		return
	}

	if err := c.DB.Create(&messaggio).Error; err != nil {
		http.Error(w, "Errore salvataggio: "+err.Error(), http.StatusInternalServerError)
		return
	}

	// Notifica in tempo reale al destinatario se online
	clientsMu.Lock()
	if destConn, online := clients[messaggio.IDDestinatario]; online {
		destConn.WriteJSON(messaggio)
	}
	clientsMu.Unlock()

	w.WriteHeader(http.StatusCreated)
	json.NewEncoder(w).Encode(messaggio)
}

// ⚡ 3. Caricamento cronologia (Al primo avvio)
func (c *MessaggioController) GetCronologiaChat(w http.ResponseWriter, r *http.Request) {

	).Order("data_invio asc").Find(&messaggi).Error

		// STAMPA L'ERRORE ESATTO NEL TERMINALE GO

		return
	}

	nuovaScheda.ID = uuid.New().String()

	// Assegnamo gli UUID, colleghiamo le chiavi esterne e verifichiamo l'esistenza dell'esercizio

			}

			nuovaScheda.Giorni[i].Esercizi[j].IDGiorno = nuovaScheda.Giorni[i].ID

			var count int64
			c.DB.Model(&models.Esercizio{}).Where("id = ?", idCercato).Count(&count)

				// Stampiamo nel terminale Go l'ID che sta fallendo la ricerca

				return
			}
		}
	}

	if risultato := c.DB.Create(&nuovaScheda); risultato.Error != nil {

		"serie_id":       idSerie,
		"esercizio_id":   serie.IDEsercizioScheda,
		"proiezione_1rm": proiezione,
	})
}

	type RisultatoVolume struct {
		GruppoMuscolare string `json:"gruppo_muscolare"`
		TotaleSerie     int    `json:"totale_serie"`
	}

	var volumi []RisultatoVolume

	// Utilizzo della query corretta con le associazioni JOIN precise
	err := c.DB.Table("serie_eseguite").
		Select("esercizios.gruppo_muscolare, count(serie_eseguite.id) as totale_serie").
		Joins("JOIN allenamentos ON allenamentos.id = serie_eseguite.allenamento_id").
		Joins("JOIN esercizios ON esercizios.id = serie_eseguite.esercizio_id").
		Where("allenamentos.id_atleta = ? AND allenamentos.data >= current_date - interval '7 days'", idAtleta).
		Group("esercizios.gruppo_muscolare").
		Scan(&volumi).Error

	if err != nil {
		http.Error(w, "Errore nel calcolo dei volumi", http.StatusInternalServerError)
		return
	}

	w.Header().Set("Content-Type", "application/json")

		"atleta_id":       idAtleta,

	})

func (c *UtenteController) GestisciCheckIn(w http.ResponseWriter, r *http.Request) {
	idStr := chi.URLParam(r, "id")
	idAtleta, _ := strconv.Atoi(idStr)

	var payload struct {
		TipoTask string `json:"tipo_task"`
		XPGained int    `json:"xp_gained"`
		Data     string `json:"data"`
	}
	json.NewDecoder(r.Body).Decode(&payload)

	dataTask, _ := time.Parse("2006-01-02", payload.Data)

	var checkin models.CheckIn
	c.DB.Where("atleta_id = ? AND data = ?", idAtleta, dataTask).FirstOrCreate(&checkin, models.CheckIn{
		AtletaID: idStr,
		Data:     dataTask,
	})

	// Toggle della spunta in base al task ricevuto
	switch payload.TipoTask {
	case "allenamento":
		checkin.AllenamentoCompletato = !checkin.AllenamentoCompletato
	case "dieta":
		checkin.DietaCompletata = !checkin.DietaCompletata
	case "cardio":
		checkin.CardioCompletato = !checkin.CardioCompletato
	}

	checkin.XPGained += payload.XPGained
	c.DB.Save(&checkin)

	// Aggiornamento livello Utente
	var utente models.Utente
	c.DB.First(&utente, idAtleta)
	utente.PuntiXP += payload.XPGained

	// Logica Level-Up
	xpNecessari := utente.Livello * 100
	if utente.PuntiXP >= xpNecessari {
		utente.PuntiXP -= xpNecessari
		utente.Livello++
	}
	c.DB.Save(&utente)

	w.Header().Set("Content-Type", "application/json")
	json.NewEncoder(w).Encode(utente)
}

package middleware

import (
	"context"
	"net/http"
	"os"
	"strings"

	"github.com/golang-jwt/jwt/v5"
)

func AuthMiddleware(next http.Handler) http.Handler {
	return http.HandlerFunc(func(w http.ResponseWriter, r *http.Request) {
		authHeader := r.Header.Get("Authorization")
		if authHeader == "" {
			http.Error(w, "Accesso Negato: Token mancante", http.StatusUnauthorized)
			return
		}

		// Il formato standard e' "Bearer <tuo_token>"
		parti := strings.Split(authHeader, " ")
		if len(parti) != 2 || parti[0] != "Bearer" {
			http.Error(w, "Accesso Negato: Formato token non valido", http.StatusUnauthorized)
			return
		}

		tokenString := parti[1]
		chiaveSegreta := os.Getenv("JWT_SECRET")

		token, err := jwt.Parse(tokenString, func(token *jwt.Token) (interface{}, error) {

			if _, ok := token.Method.(*jwt.SigningMethodHMAC); !ok {
				return nil, http.ErrAbortHandler
			}
			return []byte(chiaveSegreta), nil
		})

		if err != nil || !token.Valid {
			http.Error(w, "Accesso Negato: Token non valido o scaduto", http.StatusUnauthorized)
			return
		}

		if claims, ok := token.Claims.(jwt.MapClaims); ok && token.Valid {
			ctx := context.WithValue(r.Context(), "utente_id", claims["utente_id"])
			next.ServeHTTP(w, r.WithContext(ctx))
		} else {
			http.Error(w, "Accesso negato: Dati token illeggibili", http.StatusUnauthorized)
		}

	})
}
